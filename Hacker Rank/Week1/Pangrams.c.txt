#include <stdio.h>
#include <ctype.h>

int main(){
    char s[1001];
    int letters[26]={0};
    int i;
    
    fgets(s, sizeof(s), stdin);
    
//mark each letter found
for(i=0; s[i] != '\0'; i++){
    if(isalpha(s[i])) {
        letters[tolower(s[i])-'a']=1;
        
    }
}
//check if all 26 letters are presnt
for(i =0; i<26;i++){
    if(letters[i]==0){
        printf("not pangram\n");
        return 0;
    }
}
printf("pangram");
return 0;
}
