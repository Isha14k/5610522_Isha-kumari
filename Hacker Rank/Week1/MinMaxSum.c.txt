#include <stdio.h>
int main(){
    int arr[5];
    long total = 0;
    int i;
    
    for(i=0; i<5; i++){
        scanf("%d", &arr[i]);
        total += arr[i];
    }
    int min =arr[0], max =arr[0];
    for(i=1; i<5;i++){
        if (arr[i] < min)
           min =arr[i];
        if(arr[i] > max)
           max =arr[i];
    }
    long min_sum = total - max;
    long max_sum = total - min;
    
    printf("%ld %ld\n", min_sum, max_sum);
    return 0;
}
