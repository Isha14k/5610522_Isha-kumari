#include <stdio.h>
#include <string.h>
#include <ctype.h>

void caeserCipher(char s[],int k){
    int n=strlen(s);
    k=k %26;
    
    for(int i=0; i<n; i++){
        if(isalpha(s[i])){
            if(isupper(s[i])){
                s[i]='A'+(s[i]-'A'+k)%26;
            }else if(islower(s[i])){
                s[i]='a'+(s[i]-'a'+k)%26; 
            }
        }
    }
}
int main(){
    int n,k;
    scanf("%d",&n);
    char s[101];
    scanf("%s",s);
    scanf("%d",&k);
    caeserCipher(s,  k);
    printf("%s\n",s);
    return 0;
}
