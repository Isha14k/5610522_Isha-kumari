#include <stdio.h>
#include <stdbool.h>
 bool isPowerOfTwo(unsigned long long n){
    return (n && !(n & (n-1)));
}
unsigned long long
HighestPowerOfTwo(unsigned long long n){
    unsigned long long p= 1ULL;
    while (p<< 1ULL <= n) {
        p <<= 1ULL;
    }
    return p;
}
const char* counterGame(unsigned long long n){
    int turn =0;
    while(n>1){
        if (isPowerOfTwo(n)) {
        n >>= 1;
        }else {
        n -= HighestPowerOfTwo(n);
        }
        turn= 1-turn ;
    } return (turn == 0) ? "Richard" : "Louise";
}
int main(){
    int t;
    scanf("%d", &t);
    while (t--) {
    unsigned long long n;
    scanf("%11u", &n);
    printf("%s\n", counterGame(n));
    } 
    return 0;
}
