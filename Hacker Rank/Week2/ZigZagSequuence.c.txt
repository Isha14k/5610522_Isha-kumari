int cmpfunc(const void *a, const void *b)
{
    return(*(int*)a-*(int*)b);//for q soryt
}
void findZigZagSequence(int a[],int n)
{
    qsort(a,n, sizeof(int), cmpfunc);
    int mid =(n-1)/2;
    // swap middle with last
    int temp =a[mid];
    a[mid]=a[n-1];
    a[n-1]=temp;
    // reverse from mid+1 to n-2
    int st= mid+1;
    int ed=n-2;
    while(st <=ed){
        temp=a[st];
        a[st]=a[ed];
        a[ed]=temp;
        st++;
        ed--;
    }

}
